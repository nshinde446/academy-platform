```python
# server.py
import os
import asyncio
from typing import Optional
from mcp.server import Server, stdio_server
from mcp.types import Tool, TextContent, CallToolResult
import httpx

# Load environment
API_BASE_URL = os.getenv("ACADEMY_API_URL", "http://localhost:8000")
API_TOKEN = os.getenv("ACADEMY_API_TOKEN")

if not API_TOKEN:
    raise ValueError("ACADEMY_API_TOKEN environment variable is required")

app = Server("academy-platform")

@app.list_tools()
async def list_tools() -> list[Tool]:
    return [
        Tool(
            name="get_risk_students",
            description="Retrieves a list of students at risk in a given batch, based on attendance and test scores.",
            inputSchema={
                "type": "object",
                "properties": {
                    "batch_id": {
                        "type": "string",
                        "description": "The UUID of the batch to evaluate"
                    },
                    "threshold": {
                        "type": "number",
                        "description": "Risk threshold (0.0 to 1.0). Higher = more at-risk.",
                        "default": 0.7
                    }
                },
                "required": ["batch_id"]
            }
        )
    ]

@app.call_tool()
async def call_tool(name: str, arguments: dict) -> list[TextContent]:
    if name != "get_risk_students":
        raise ValueError(f"Unknown tool: {name}")

    batch_id = arguments.get("batch_id")
    threshold = arguments.get("threshold", 0.7)

    # Build the backend URL
    url = f"{API_BASE_URL}/api/v1/analytics/batches/{batch_id}/risk-students"
    headers = {"Authorization": f"Bearer {API_TOKEN}"}
    params = {"threshold": threshold}

    async with httpx.AsyncClient(timeout=30.0) as client:
        response = await client.get(url, headers=headers, params=params)

    if response.status_code != 200:
        error_msg = f"Backend error: {response.status_code} - {response.text}"
        return [TextContent(type="text", text=error_msg)]

    data = response.json()

    if not data:
        return [TextContent(type="text", text="No at‑risk students found for this batch.")]

    # Format a readable summary
    lines = [f"## At‑Risk Students (threshold: {threshold:.2f})"]
    lines.append(f"Batch ID: `{batch_id}`")
    lines.append("")
    
    for student in data:
        name = student.get("name", "Unknown")
        risk_score = student.get("risk_score", 0.0)
        attendance = student.get("attendance_percentage", 0.0)
        avg_score = student.get("average_score", 0.0)
        weak_topics = student.get("weak_topics", [])
        
        lines.append(f"### {name}")
        lines.append(f"- **Risk score:** {risk_score:.2f}")
        lines.append(f"- **Attendance:** {attendance:.1f}%")
        lines.append(f"- **Average score:** {avg_score:.1f}%")
        if weak_topics:
            topics_str = ", ".join(weak_topics[:3])
            lines.append(f"- **Weak topics:** {topics_str}")
        lines.append("")

    return [TextContent(type="text", text="\n".join(lines))]

async def main():
    async with stdio_server() as (read_stream, write_stream):
        await app.run(
            read_stream,
            write_stream,
            app.create_initialization_options()
        )

if __name__ == "__main__":
    asyncio.run(main())
```

## 🚀 Step 1 – Create the Server File

```python
# mcp_academy_server.py
import os
import asyncio
import base64
from typing import Optional
from mcp.server import Server, stdio_server
from mcp.types import Tool, TextContent, CallToolResult
import httpx

# Load environment
API_BASE_URL = os.getenv("ACADEMY_API_URL", "http://localhost:8000")
API_TOKEN = os.getenv("ACADEMY_API_TOKEN")
if not API_TOKEN:
    raise ValueError("ACADEMY_API_TOKEN is required")

app = Server("academy-platform")

async def _request(method: str, endpoint: str, **kwargs):
    """Helper: make authenticated API call."""
    headers = {"Authorization": f"Bearer {API_TOKEN}"}
    if kwargs.get("json"):
        headers["Content-Type"] = "application/json"
    async with httpx.AsyncClient(timeout=30.0) as client:
        url = f"{API_BASE_URL}{endpoint}"
        resp = await client.request(method, url, headers=headers, **kwargs)
        if resp.status_code >= 400:
            raise Exception(f"API error {resp.status_code}: {resp.text}")
        return resp.json()

@app.list_tools()
async def list_tools() -> list[Tool]:
    return [
        Tool(
            name="get_risk_students",
            description="Retrieves at‑risk students for a batch.",
            inputSchema={
                "type": "object",
                "properties": {
                    "batch_id": {"type": "string", "description": "UUID of the batch"},
                    "threshold": {"type": "number", "description": "Risk threshold (0-1)", "default": 0.7}
                },
                "required": ["batch_id"]
            }
        ),
        Tool(
            name="import_syllabus",
            description="Imports an Excel syllabus file into a course. Provide file content as base64.",
            inputSchema={
                "type": "object",
                "properties": {
                    "course_id": {"type": "string", "description": "UUID of the course"},
                    "file_data_base64": {"type": "string", "description": "Base64‑encoded .xlsx file content"},
                    "filename": {"type": "string", "description": "Original filename (optional)"}
                },
                "required": ["course_id", "file_data_base64"]
            }
        ),
        Tool(
            name="create_lecture",
            description="Creates a new lecture for a batch and teacher.",
            inputSchema={
                "type": "object",
                "properties": {
                    "batch_id": {"type": "string", "description": "UUID of the batch"},
                    "teacher_id": {"type": "string", "description": "UUID of the teacher"},
                    "subject_id": {"type": "string", "description": "UUID of the subject"},
                    "topic_id": {"type": "string", "description": "UUID of the topic (optional)"},
                    "scheduled_start": {"type": "string", "description": "ISO datetime (e.g., 2026-05-20T09:00:00)"},
                    "duration_minutes": {"type": "integer", "description": "Lecture duration in minutes", "default": 60},
                    "delivery_mode": {"type": "string", "description": "offline|online|hybrid", "default": "offline"}
                },
                "required": ["batch_id", "teacher_id", "subject_id", "scheduled_start"]
            }
        ),
        Tool(
            name="get_student_attendance",
            description="Returns attendance summary for a student.",
            inputSchema={
                "type": "object",
                "properties": {
                    "student_id": {"type": "string", "description": "UUID of the student"}
                },
                "required": ["student_id"]
            }
        ),
        Tool(
            name="get_batch_analytics",
            description="Returns key metrics for a batch (attendance %, avg score, top performers, weak subjects).",
            inputSchema={
                "type": "object",
                "properties": {
                    "batch_id": {"type": "string", "description": "UUID of the batch"}
                },
                "required": ["batch_id"]
            }
        ),
        Tool(
            name="get_teacher_productivity",
            description="Returns lecture completion metrics for a teacher.",
            inputSchema={
                "type": "object",
                "properties": {
                    "teacher_id": {"type": "string", "description": "UUID of the teacher"}
                },
                "required": ["teacher_id"]
            }
        )
    ]

@app.call_tool()
async def call_tool(name: str, arguments: dict) -> list[TextContent]:
    if name == "get_risk_students":
        batch_id = arguments["batch_id"]
        threshold = arguments.get("threshold", 0.7)
        data = await _request("GET", f"/api/v1/analytics/batches/{batch_id}/risk-students", params={"threshold": threshold})
        if not data:
            return [TextContent(type="text", text="No at‑risk students found for this batch.")]
        lines = [f"## At‑Risk Students (threshold {threshold:.2f})", f"Batch ID: `{batch_id}`", ""]
        for s in data:
            lines.append(f"- {s.get('name', 'Unknown')}: risk {s.get('risk_score',0):.2f}, attendance {s.get('attendance_percentage',0):.1f}%, avg score {s.get('average_score',0):.1f}%")
        return [TextContent(type="text", text="\n".join(lines))]

    elif name == "import_syllabus":
        course_id = arguments["course_id"]
        file_b64 = arguments["file_data_base64"]
        filename = arguments.get("filename", "upload.xlsx")
        file_bytes = base64.b64decode(file_b64)
        files = {"file": (filename, file_bytes, "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet")}
        result = await _request("POST", f"/api/v1/syllabus/import?course_id={course_id}", files=files)
        return [TextContent(type="text", text=f"✅ Syllabus import successful.\n{result}")]

    elif name == "create_lecture":
        payload = {
            "batch_id": arguments["batch_id"],
            "teacher_id": arguments["teacher_id"],
            "subject_id": arguments["subject_id"],
            "topic_id": arguments.get("topic_id"),
            "scheduled_start": arguments["scheduled_start"],
            "duration_minutes": arguments.get("duration_minutes", 60),
            "delivery_mode": arguments.get("delivery_mode", "offline")
        }
        result = await _request("POST", "/api/v1/lectures", json=payload)
        return [TextContent(type="text", text=f"✅ Lecture created successfully.\nID: {result.get('id', 'unknown')}")]

    elif name == "get_student_attendance":
        student_id = arguments["student_id"]
        data = await _request("GET", f"/api/v1/attendance/student/{student_id}")
        lines = [f"## Attendance for Student `{student_id}`"]
        if not data:
            lines.append("No attendance records found.")
        else:
            total = len(data)
            present = sum(1 for r in data if r.get('status') == 'PRESENT')
            absent = sum(1 for r in data if r.get('status') == 'ABSENT')
            late = sum(1 for r in data if r.get('status') == 'LATE')
            lines.append(f"- Total lectures: {total}")
            lines.append(f"- Present: {present} ({present/total*100:.1f}%)")
            lines.append(f"- Absent: {absent} ({absent/total*100:.1f}%)")
            lines.append(f"- Late: {late} ({late/total*100:.1f}%)")
        return [TextContent(type="text", text="\n".join(lines))]

    elif name == "get_batch_analytics":
        batch_id = arguments["batch_id"]
        data = await _request("GET", f"/api/v1/analytics/batches/{batch_id}")
        lines = [f"## Batch Analytics for `{batch_id}`"]
        lines.append(f"- Total students: {data.get('total_students', 'N/A')}")
        lines.append(f"- Average attendance: {data.get('avg_attendance', 'N/A'):.1f}%")
        lines.append(f"- Average score: {data.get('avg_score', 'N/A'):.1f}%")
        top = data.get('top_performers', [])
        if top:
            lines.append(f"- Top performers: {', '.join(top[:3])}")
        weak = data.get('weak_subjects', [])
        if weak:
            lines.append(f"- Weak subjects: {', '.join(weak[:3])}")
        return [TextContent(type="text", text="\n".join(lines))]

    elif name == "get_teacher_productivity":
        teacher_id = arguments["teacher_id"]
        data = await _request("GET", f"/api/v1/analytics/teachers/{teacher_id}")
        lines = [f"## Teacher Productivity for `{teacher_id}`"]
        lines.append(f"- Total lectures: {data.get('total_lectures', 'N/A')}")
        lines.append(f"- Completed: {data.get('completed_lectures', 'N/A')}")
        lines.append(f"- Completion rate: {data.get('completion_rate', 'N/A'):.1f}%")
        lines.append(f"- Average delay: {data.get('avg_delay_minutes', 'N/A'):.1f} minutes")
        lines.append(f"- Average student attendance: {data.get('avg_student_attendance', 'N/A'):.1f}%")
        return [TextContent(type="text", text="\n".join(lines))]

    else:
        raise ValueError(f"Unknown tool: {name}")

async def main():
    async with stdio_server() as (read_stream, write_stream):
        await app.run(read_stream, write_stream, app.create_initialization_options())

if __name__ == "__main__":
    asyncio.run(main())
```
